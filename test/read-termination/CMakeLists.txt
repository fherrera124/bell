cmake_minimum_required(VERSION 3.16)
project(bell_read_termination_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(BELL_DISABLE_TESTS ON CACHE BOOL "" FORCE)
set(BELL_DISABLE_CODECS ON CACHE BOOL "" FORCE)
set(BELL_DISABLE_CONTAINERS ON CACHE BOOL "" FORCE)
set(BELL_DISABLE_MDNS ON CACHE BOOL "" FORCE)
set(BELL_DISABLE_MQTT ON CACHE BOOL "" FORCE)
set(BELL_DISABLE_TAOJSON ON CACHE BOOL "" FORCE)
find_path(DOCTEST_INCLUDE_DIR doctest/doctest.h
          HINTS "${CMAKE_CURRENT_SOURCE_DIR}/../../external/doctest" REQUIRED)
add_subdirectory(../.. bell)
enable_testing()
add_executable(bell-read-tests main.cpp TCPReadTest.cpp ../HTTPReadTerminationTest.cpp ../HTTPReaderTest.cpp)
target_include_directories(bell-read-tests PRIVATE "${DOCTEST_INCLUDE_DIR}")
target_link_libraries(bell-read-tests PRIVATE bell)
target_compile_options(bell-read-tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME bell-read-tests COMMAND bell-read-tests)
if(NOT BELL_DISABLE_MBEDTLS)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  find_program(OPENSSL_EXECUTABLE openssl REQUIRED)
  add_executable(bell-tls-read-probe TLSReadProbe.cpp)
  target_link_libraries(bell-tls-read-probe PRIVATE bell)
  add_test(NAME bell-tls-read-tests COMMAND "${Python3_EXECUTABLE}"
           "${CMAKE_CURRENT_SOURCE_DIR}/tls_read_test.py"
           "$<TARGET_FILE:bell-tls-read-probe>" "${OPENSSL_EXECUTABLE}")
  set_tests_properties(bell-tls-read-tests PROPERTIES TIMEOUT 45)
endif()
